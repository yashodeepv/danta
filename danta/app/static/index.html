# System: order-api
REST API for placing orders.

## Flow: place_order
Trigger: POST /orders with
   order_id str, email str, amount int
Mode: sync
Steps:
1. Validate: `amount > 0`. Otherwise
   stop with error "invalid amount".
2. Call payments.charge with
   order_id, amount.
3. Respond with order_id, status.
compiled → 
@router.post("/orders")
async def place_order(
    order_id: str,
    email: str,
    amount: int,
):
    if not (amount > 0):
        raise HTTPException(400,
            detail="invalid amount")
    charge = await payments.charge(
        order_id=order_id, amount=amount)
    return {"order_id": order_id,
            "status": "ok"}
router.post("/orders", async (req, res) => {
  const { order_id, email, amount } = req.body;
  if (!(amount > 0)) {
    res.status(400).json({
      error: "invalid amount" });
    return;
  }
  const charge = await payments.charge({
    order_id, amount });
  res.json({ order_id, status: "ok" });
});
@PostMapping("/orders")
public ResponseEntity<?> placeOrder(
    @RequestBody OrderRequest req) {
  if (!(req.getAmount() > 0)) {
    return ResponseEntity.badRequest()
      .body(Map.of("error",
            "invalid amount"));
  }
  payments.charge(req.getOrderId(),
                  req.getAmount());
  return ResponseEntity.ok(Map.of(
    "order_id", req.getOrderId(),
    "status", "ok"));
}
