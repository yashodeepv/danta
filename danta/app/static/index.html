# System: shop
Online shop with orders and inventory.

## Flow: place_order
Trigger: POST /orders with
   customer_email str, total int
Steps:
1. Validate: `total > 0`.
2. Save order with customer_email, total,
   status "pending".
3. Email customer_email: "Order confirmed."
4. Respond with order_id, status.

@router.post("/orders")
async def place_order(
    customer_email: str,
    total: int,
):
    if not (total > 0):
        raise HTTPException(400)
    order_id = store.save("order", {
        "customer_email": customer_email,
        "total": total,
        "status": "pending"})
    email(customer_email,
          "Order confirmed.")
    return {"order_id": order_id,
            "status": "ok"}
router.post("/orders", async (req, res) => {
  const { customer_email, total } = req.body;
  if (!(total > 0))
    return res.status(400).json({});
  const order_id = store.save("order", {
    customer_email, total,
    status: "pending" });
  await email(customer_email,
              "Order confirmed.");
  res.json({ order_id, status: "ok" });
});
@PostMapping("/orders")
public ResponseEntity<?> placeOrder(
    @RequestBody OrderRequest req) {
  if (!(req.getTotal() > 0))
    return ResponseEntity
        .badRequest().build();
  var id = store.save("order", Map.of(
    "customer_email", req.getEmail(),
    "total", req.getTotal(),
    "status", "pending"));
  email(req.getEmail(),
        "Order confirmed.");
  return ResponseEntity.ok(
    Map.of("order_id", id));
}
